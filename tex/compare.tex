% The Mills page side by side: the 1947 scan and Mills 8A (pdfLaTeX).
% render.sh builds this (pdflatex, in out/) and joins it with the full-size
% pages (compare-scan.tex, mills-8a.pdf) into out/mills-compare.pdf.
%
% Each panel is its page cropped to the ink (bbox.tex, written by render.sh;
% the scan is cropped already) and set at the same width, so the titles, the
% text blocks and the page numbers line up across the two.
\documentclass{article}
\usepackage[paperwidth=62pc,paperheight=52pc,margin=1pc]{geometry}
\usepackage{graphicx}
\pagestyle{empty}
\setlength\parindent{0pt}
\input{bbox}
\newcommand\cropped[2]{% page #2 cut to the ink box #1
  \edef\next{\noexpand\includegraphics[viewport=#1,clip,width=26pc]{#2}}\next}
\newcommand\panel[2]{%
  \begin{minipage}[t][48pc][t]{29pc}\centering
    \vspace{1.5pc}#1\par\vfill{\small\sffamily #2}
  \end{minipage}}
\begin{document}
\centering
\panel{\includegraphics[width=26pc]{../scan-mills1947.png}}
  {1947 \textit{Bulletin of the AMS} (scan)}\hfill
\panel{\cropped\bboxpdf{mills-8a.pdf}}
  {Mills 8A (\texttt{\string\usepackage\string{mills8a\string}}, pdfLaTeX)}
\end{document}
